Optional properties, exactOptionalPropertyTypes, indexed access, noUncheckedIndexedAccess, and safe absence

TypeScript checks information about JavaScript code before the JavaScript program runs.

Senior practice for this track ↗ · The reading estimate excludes the lab and interview rehearsal.

An omitted property and a present property containing undefined are different object shapes. exactOptionalPropertyTypes preserves that distinction when assigning optional fields. noUncheckedIndexedAccess makes uncertain dictionary reads include undefined. These flags add static checks; they do not validate incoming JSON.

An update form can leave a field untouched, supply a replacement, or explicitly request clearing it. These are three instructions, not one truthy-or-falsy choice.

01

Optional properties

An optional property may be absent. With exactOptionalPropertyTypes, explicitly assigning undefined is different unless undefined is part of the value type.

02

exactOptionalPropertyTypes

exactOptionalPropertyTypes distinguishes an absent optional property from a present property containing undefined, unless undefined is explicitly part of that property's value type.

03

indexed access

An indexed read such as counts[key] can encounter a missing key. With noUncheckedIndexedAccess, an uncertain dictionary read includes undefined; check it before treating the result as a number.

04

noUncheckedIndexedAccess

noUncheckedIndexedAccess adds undefined to uncertain array and index-signature reads, exposing that a syntactically valid key may be missing at runtime.

05

safe absence

Give absence an explicit meaning. In a patch, an omitted name can mean leave unchanged, while a present name requests replacement. Do not use truthiness to detect presence: an empty string and zero can be legitimate supplied values.

// Save as optional.ts, then run (the two extra flags are not enabled by strict):
// tsc --strict --exactOptionalPropertyTypes --noUncheckedIndexedAccess --target ES2022 --outDir out optional.ts
// node out/optional.js
type Profile = { name: string | null };
type NamePatch = { name?: string | null }; // omitted: keep; string: set; null: clear

function check(condition: boolean): void {
  if (!condition) throw new Error("check failed");
}

// This parser accepts decoded JSON, with an explicit unknown-field policy.
function parsePatch(input: unknown): NamePatch {
  if (typeof input !== "object" || input === null || Array.isArray(input)) {
    throw new TypeError("patch object required");
  }
  if (Object.keys(input).some(key => key !== "name")) {
    throw new TypeError("unknown patch field");
  }
  if (!Object.hasOwn(input, "name")) return {};
  const value = input as Record<string, unknown>;
  const name = value.name;
  if (name !== null && typeof name !== "string") {
    throw new TypeError("name must be a string or null");
  }
  return { name };
}

// Call with the normalized output of parsePatch, not an unchecked assertion.
function applyPatch(current: Profile, patch: NamePatch): Profile {
  if ("name" in patch) return { ...current, name: patch.name };
  return current;
}

const current: Profile = { name: "Ada" };
for (const [json, expected] of [
  ['{}', "Ada"],
  ['{"name":"Grace"}', "Grace"],
  ['{"name":""}', ""], // empty is a supplied string, not absence
  ['{"name":null}', null]
] as const) {
  check(applyPatch(current, parsePatch(JSON.parse(json))).name === expected);
}
check(current.name === "Ada"); // updates did not mutate the original
for (const invalid of [null, [], { name: undefined }, { name: 7 }, { admin: true }]) {
  let rejected = false;
  try { parsePatch(invalid); } catch (error) {
    if (!(error instanceof TypeError)) throw error;
    rejected = true;
  }
  check(rejected);
}

const counts: Record<string, number> = { empty: 0 };
function countOrDefault(key: string): number { return counts[key] ?? 10; }
check(countOrDefault("empty") === 0);
check(countOrDefault("missing") === 10);
check(JSON.stringify({ name: undefined }) === "{}"); // undefined is not a JSON value
check(JSON.stringify({ name: null }) === '{"name":null}');

function compilerExamples(key: string): void {
  // @ts-expect-error a present undefined is outside NamePatch's declared contract
  const invalidPatch: NamePatch = { name: undefined };
  // @ts-expect-error an arbitrary key can be missing
  const count: number = counts[key];
}
