Runtime validation

Runtime validation examines actual external values after types have erased and produces trusted data or structured errors.

unknown input

unknown can contain any runtime value but permits no unsafe operation until control flow or parsing supplies evidence of a narrower type.

parsing

Parsing should accept unknown input and return a domain representation only after checking shape, values, limits, versions, and invariants.

schemas

A runtime schema is executable validation metadata that can sometimes also infer a static type. The two surfaces must be kept from drifting.

decoders

A decoder checks an external representation and returns a validated value or explicit errors, rather than trusting a cast.

trusted domain types

Domain types represent values accepted under stated checks. Trust is scoped to those checks, not automatic permission or permanent immutability.

type Result<T, E> =
  | { ok: true; value: T }
  | { ok: false; errors: E[] };

type CreateProject = { name: string; ownerId: string };

function parseCreateProject(input: unknown): Result<CreateProject, string> {
  if (typeof input !== "object" || input === null || Array.isArray(input)) {
    return { ok: false, errors: ["body must be an object"] };
  }
  const value = input as Record<string, unknown>;
  const errors = [] as string[];
  if (typeof value.name !== "string" || !value.name.trim()) errors.push("name required");
  if (typeof value.ownerId !== "string") errors.push("ownerId required");
  if (errors.length) return { ok: false, errors };
  return { ok: true, value: { name: (value.name as string).trim(), ownerId: value.ownerId as string } };
}

function check(condition: boolean): void {
  if (!condition) throw new Error("check failed");
}
const command = parseCreateProject({ name: " Demo ", ownerId: "u1" });
check(command.ok && command.value.name === "Demo");
for (const invalid of [null, [], {}, { name: " ", ownerId: "u1" }, { name: "ok", ownerId: 7 }]) {
  check(!parseCreateProject(invalid).ok);
}
console.log("Validation checks passed");
