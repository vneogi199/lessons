01

Process lifecycle

A production Node process initializes resources, becomes ready, admits bounded work, stops admission on termination, drains, cleans up, and exits under a deadline.

02

signals

A signal notifies a process of an event such as termination. Some can be handled for cleanup; SIGKILL cannot be caught or used for graceful shutdown.

03

readiness

Readiness answers whether this instance should receive new traffic. It should fall before draining starts.

04

liveness

Liveness answers whether a stuck instance needs replacement. It should not fail merely because a temporary dependency is unavailable.

05

graceful shutdown

Graceful shutdown stops new work, allows bounded completion, cancels or closes remaining resources, and exits before the platform's termination deadline.

06

draining

Draining stops new work and allows bounded existing work to finish or be safely released before shutdown.

07

deadlines

A deadline is the latest acceptable completion time. Each downstream operation should consume the remaining budget rather than starting a fresh full timeout.

import assert from "node:assert/strict";

// Node HTTP shutdown coordinator. Tests below use fakes, not listening sockets.
function createShutdown(server, controller, setReady, deadlineMs = 10000) {
  let pending;
  return function shutdown() {
    if (pending) return pending; // Repeated signals share one drain operation.
    setReady(false);
    pending = new Promise((resolve, reject) => {
      const timer = setTimeout(() => {
        const error = new Error("drain deadline");
        controller.abort(error);
        server.closeAllConnections();
        reject(error);
      }, deadlineMs);
      server.close(error => {
        clearTimeout(timer);
        if (error) reject(error);
        else resolve();
      });
    });
    return pending;
  };
}

let closes = 0, ready = true;
const server = {
  close(done) { closes++; queueMicrotask(() => done()); },
  closeAllConnections() { throw new Error("successful drain must not force close"); }
};
const controller = new AbortController();
const shutdown = createShutdown(server, controller, value => { ready = value; });
const first = shutdown();
assert.equal(shutdown(), first);
await first;
assert.equal(closes, 1);
assert.equal(ready, false);
assert.equal(controller.signal.aborted, false);

let forced = false;
const stuckController = new AbortController();
const stuck = createShutdown({
  close() {},
  closeAllConnections() { forced = true; }
}, stuckController, () => {}, 1);
await assert.rejects(stuck(), /drain deadline/);
assert.equal(forced, true);
assert.equal(stuckController.signal.aborted, true);
